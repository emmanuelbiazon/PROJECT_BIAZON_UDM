<?php

}

$sql = "SELECT 
            t.tnum AS teacherID,
            t.tlname AS teacher_lastname,
            s.studentID,
            s.studentlastname
        FROM enrollment e
        JOIN teachers t ON e.tnum = t.tnum
        JOIN students s ON e.studentID = s.studentID
        WHERE e.subjectID = 'ICT01'
        ORDER BY s.studentlastname ASC";

$result = mysqli_query($conn, $sql);
